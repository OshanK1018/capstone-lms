# login as admin and capture jwt token
token=$(curl -s -X POST http://localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"tkhandaker@hunter.cuny.edu","password":"passadmin"}' \
  | jq -r '.jwtToken')

echo "token: $token"
