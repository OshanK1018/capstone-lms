# login as admin and capture jwt token
token=$(curl -s -X POST http://localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"tkhandaker@hunter.cuny.edu","password":"passadmin"}' \
  | jq -r '.jwtToken')

echo "token: $token"
echo

# get current user (authenticated)
curl -s -X GET http://localhost:3000/api/auth/me \
  -H "Authorization: Bearer $token" \
  | jq .

# get user by id (no auth required)
curl -s -X GET http://localhost:3000/api/Users/4 \

# create a course using term (admin can create for any instructor)

  -d '{"title":"CSCI265 – Web Dev","term":"Fall 2026","instructor_id":2,"max_seats":30,"credits":3,"materials_url":null}' \
  | jq .
